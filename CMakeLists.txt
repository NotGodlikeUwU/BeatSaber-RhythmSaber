cmake_minimum_required(VERSION 3.22)
option(RHYTHMSABER_HOST_TESTS "Build portable math tests only" OFF)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(RHYTHMSABER_HOST_TESTS)
  project(RhythmSaberMath LANGUAGES CXX)
  enable_testing()
  add_executable(timing_tests tests/timing_tests.cpp src/TimingScoring.cpp src/GuardPolicy.cpp src/MapCompatibility.cpp src/TrajectoryTiming.cpp)
  target_include_directories(timing_tests PRIVATE include extern/includes)
  add_test(NAME timing_math COMMAND timing_tests)
  return()
endif()
include(cmake/targets/android-ndk.cmake)
file(READ "${CMAKE_CURRENT_SOURCE_DIR}/qpm.json" RHYTHMSABER_PACKAGE_JSON)
string(JSON RHYTHMSABER_VERSION GET "${RHYTHMSABER_PACKAGE_JSON}" info version)
project(rhythmsaber VERSION ${RHYTHMSABER_VERSION} LANGUAGES C CXX)
set(COMPILE_ID rhythmsaber)
set(EXTERN_DIR ${CMAKE_CURRENT_SOURCE_DIR}/extern)
include(cmake/utils.cmake)
add_library(rhythmsaber SHARED src/main.cpp src/Hooks.cpp src/TimingScoring.cpp src/TrajectoryTiming.cpp src/TimingManager.cpp src/GuardPolicy.cpp src/MapCompatibility.cpp
  src/JudgementLine.cpp src/TimingFeedback.cpp src/ScoreSubmissionGuard.cpp src/RhythmSaberConfig.cpp src/RhythmSaberUI.cpp)
target_include_directories(rhythmsaber PRIVATE include shared)
target_compile_definitions(rhythmsaber PRIVATE MOD_ID="rhythmsaber" VERSION="${PROJECT_VERSION}" QUEST)
target_compile_options(rhythmsaber PRIVATE -frtti -fexceptions -fvisibility=hidden -fPIC -Wall -Wextra -Wno-unused-parameter -Wno-unused-function -Wno-unneeded-internal-declaration)
include(extern.cmake)
target_link_libraries(rhythmsaber PRIVATE log)
target_link_options(rhythmsaber PRIVATE -Wl,--no-undefined)
file(GLOB INLINE_C ${EXTERN_DIR}/includes/beatsaber-hook/shared/inline-hook/*.c)
file(GLOB INLINE_CPP ${EXTERN_DIR}/includes/beatsaber-hook/shared/inline-hook/*.cpp)
target_sources(rhythmsaber PRIVATE ${INLINE_C} ${INLINE_CPP})
add_custom_command(TARGET rhythmsaber POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy $<TARGET_FILE:rhythmsaber> ${CMAKE_CURRENT_BINARY_DIR}/librhythmsaber.debug.so
  COMMAND ${CMAKE_STRIP} --strip-debug $<TARGET_FILE:rhythmsaber>)
